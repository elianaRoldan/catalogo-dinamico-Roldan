Código Backend para Apps Script (doGet)

En tu hoja de cálculo ve a Extensiones → Apps Script, pega este código y despliega como Aplicación Web con acceso a Cualquier usuario (Anyone):

function doGet(e) { try { // 1. Abrir la hoja activa y buscar la pestaña "Servicios" var libro = SpreadsheetApp.getActiveSpreadsheet(); var hoja = libro.getSheetByName("Servicios"); if (!hoja) { return ContentService.createTextOutput(JSON.stringify({ error: "No se encontró la hoja llamada 'Servicios'." })).setMimeType(ContentService.MimeType.JSON); } // 2. Obtener todos los datos de la hoja var datos = hoja.getDataRange().getValues(); // Fila 1 son los encabezados: Nombre, Precio, Descripcion, Imagen // A partir de la fila 2 están los 6 servicios var resultado = []; for (var i = 1; i < datos.length; i++) { var fila = datos[i]; if (!fila[0]) continue; // Ignorar filas vacías resultado.push({ nombre: fila[0] ? String(fila[0]).trim() : "", precio: Number(fila[1]) || 0, descripcion: fila[2] ? String(fila[2]).trim() : "", imagen: fila[3] ? String(fila[3]).trim() : "" }); } // 3. Devolver los datos en formato JSON para que el frontend los lea return ContentService.createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ error: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
Pasos para desplegar:
1. Clic en botón azul Implementar → Nueva implementación.
2. Seleccionar tipo Aplicación web.
3. En Quién tiene acceso, seleccionar obligatoriamente: Cualquier usuario (Anyone).
4. Copiar la URL que termina en /exec y pegarla en el campo superior de esta web.
